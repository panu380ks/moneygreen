//@version=6
strategy("MOST Strategy", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=100)

// Inputs
length = input.int(8, "Length")
percent = input.float(2.0, "Percent")
atrLen = input.int(14, "ATR Length")
atrMult = input.float(1.5, "ATR Multiplier")
trendLen = input.int(200, "Trend EMA Length")
startDate = input.time(timestamp("1 Jan 2018 00:00 +0000"), "Start Date")
endDate = input.time(timestamp("31 Dec 2069 23:59 +0000"), "End Date")

// Date Filter
timeOK = time >= startDate and time <= endDate

// Trend Filter
trendEma = ta.ema(close, trendLen)
uptrend = close > trendEma
downtrend = close < trendEma

// MOST Calculation
exMov = ta.ema(close, length)
fark = exMov * percent * 0.01
longStop = exMov - fark
longStopPrev = nz(longStop[1], longStop)
longStop := exMov > longStopPrev ? math.max(longStop, longStopPrev) : longStop
shortStop = exMov + fark
shortStopPrev = nz(shortStop[1], shortStop)
shortStop := exMov < shortStopPrev ? math.min(shortStop, shortStopPrev) : shortStop
var int dir = 1
dir := dir == -1 and exMov > shortStopPrev ? 1 : dir == 1 and exMov < longStopPrev ? -1 : dir
MOST = dir == 1 ? longStop : shortStop

// Signals - only buy in uptrend, only sell in downtrend
buySignal = timeOK and dir == 1 and dir[1] == -1 and uptrend
sellSignal = timeOK and dir == -1 and dir[1] == 1 and downtrend

// ATR Stop Loss
atrValue = ta.atr(atrLen)

// Entries & Exits
if buySignal
    strategy.entry("Long", strategy.long)
if sellSignal
    strategy.entry("Short", strategy.short)
if strategy.position_size > 0
    stopPrice = strategy.position_avg_price - atrMult * atrValue
    strategy.exit("ATR SL Long", "Long", stop=stopPrice)
if strategy.position_size < 0
    stopPrice = strategy.position_avg_price + atrMult * atrValue
    strategy.exit("ATR SL Short", "Short", stop=stopPrice)
if dir == -1 and strategy.position_size > 0
    strategy.close("Long")
if dir == 1 and strategy.position_size < 0
    strategy.close("Short")

// Plots
plot(trendEma, "Trend EMA", color=color.white, linewidth=1)
plot(MOST, "MOST", color=dir == 1 ? color.green : color.red, linewidth=2)
plot(exMov, "exMov", color=color.orange)
plotshape(buySignal, "Buy", shape.triangleup, location.belowbar, color.green, size=size.small)
plotshape(sellSignal, "Sell", shape.triangledown, location.abovebar, color.red, size=size.small)
