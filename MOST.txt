//@version=6
strategy("BB + MOST Combined Strategy", overlay=true, default_qty_type=strategy.percent_of_equity, default_qty_value=100)

// ═══════════════════════════════════════════════════
// INPUTS
// ═══════════════════════════════════════════════════

// Bollinger Bands
bbLength    = input.int(20, "BB Length", minval=1, group="Bollinger Bands")
bbMult      = input.float(2.0, "BB Multiplier", minval=0.001, maxval=50, group="Bollinger Bands")

// MOST
mostLength  = input.int(8, "MOST EMA Length", group="MOST")
mostPercent = input.float(2.0, "MOST Percent", group="MOST")

// Trend
trendLen    = input.int(200, "Trend EMA Length", group="Trend Filter")

// ATR
atrLen      = input.int(14, "ATR Length", group="Risk Management")

// Long Risk
slMultLong  = input.float(1.5, "Long Stop Loss ATR", minval=1.0, maxval=5.0, step=0.5, group="Long Risk")
tpMultLong  = input.float(3.0, "Long Take Profit ATR", minval=1.0, maxval=10.0, step=0.5, group="Long Risk")

// Short Risk
slMultShort = input.float(1.5, "Short Stop Loss ATR", minval=1.0, maxval=5.0, step=0.5, group="Short Risk")
tpMultShort = input.float(3.0, "Short Take Profit ATR", minval=1.0, maxval=10.0, step=0.5, group="Short Risk")

// Time
entryHours  = input.int(8, "Entry Window (hours from session open)", minval=1, maxval=24, group="Time Filter")
startDate   = input.time(timestamp("1 Jan 2024 00:00 +0000"), "Start Date", group="Time Filter")
endDate     = input.time(timestamp("31 Dec 2069 23:59 +0000"), "End Date", group="Time Filter")

// Direction
direction   = input.int(0, "Strategy Direction (0=Both, 1=Long, -1=Short)", minval=-1, maxval=1, group="Direction")
strategy.risk.allow_entry_in(direction == 0 ? strategy.direction.all : (direction < 0 ? strategy.direction.short : strategy.direction.long))

// ═══════════════════════════════════════════════════
// TIME FILTERS
// ═══════════════════════════════════════════════════
timeOK = time >= startDate and time <= endDate
msSinceOpen = time - time("D")
entryWindowOK = msSinceOpen >= 0 and msSinceOpen < entryHours * 60 * 60 * 1000

// ═══════════════════════════════════════════════════
// TREND FILTER (200 EMA)
// ═══════════════════════════════════════════════════
trendEma  = ta.ema(close, trendLen)
uptrend   = close > trendEma
downtrend = close < trendEma

// ═══════════════════════════════════════════════════
// BOLLINGER BANDS
// ═══════════════════════════════════════════════════
basis = ta.sma(close, bbLength)
dev   = bbMult * ta.stdev(close, bbLength)
upper = basis + dev
lower = basis - dev

bbBuySignal  = ta.crossover(close, lower)
bbSellSignal = ta.crossunder(close, upper)

// ═══════════════════════════════════════════════════
// MOST (Moving Stop)
// ═══════════════════════════════════════════════════
exMov = ta.ema(close, mostLength)
fark  = exMov * mostPercent * 0.01

longStop     = exMov - fark
longStopPrev = nz(longStop[1], longStop)
longStop    := exMov > longStopPrev ? math.max(longStop, longStopPrev) : longStop

shortStop     = exMov + fark
shortStopPrev = nz(shortStop[1], shortStop)
shortStop    := exMov < shortStopPrev ? math.min(shortStop, shortStopPrev) : shortStop

var int dir = 1
dir := dir == -1 and exMov > shortStopPrev ? 1 : dir == 1 and exMov < longStopPrev ? -1 : dir
MOST = dir == 1 ? longStop : shortStop

mostBullish = dir == 1
mostBearish = dir == -1
mostFlipLong  = dir == 1 and dir[1] == -1
mostFlipShort = dir == -1 and dir[1] == 1

// ═══════════════════════════════════════════════════
// ATR
// ═══════════════════════════════════════════════════
atrValue = ta.atr(atrLen)

// ═══════════════════════════════════════════════════
// COMBINED ENTRY SIGNALS (only during 8hr window)
// ═══════════════════════════════════════════════════
longEntry  = timeOK and entryWindowOK and bbBuySignal  and mostBullish and uptrend
shortEntry = timeOK and entryWindowOK and bbSellSignal and mostBearish and downtrend

mostLongEntry  = timeOK and entryWindowOK and mostFlipLong  and uptrend   and close <= basis
mostShortEntry = timeOK and entryWindowOK and mostFlipShort and downtrend and close >= basis

buySignal  = longEntry  or mostLongEntry
sellSignal = shortEntry or mostShortEntry

// ═══════════════════════════════════════════════════
// ENTRIES (restricted to 8hr window)
// ═══════════════════════════════════════════════════
if buySignal
    strategy.entry("Long", strategy.long)
if sellSignal
    strategy.entry("Short", strategy.short)

// ═══════════════════════════════════════════════════
// EXITS — run anytime, no time restriction
// Position stays open past the 8hr window.
// Closes when first hit: TP, SL, or MOST reversal.
// ═══════════════════════════════════════════════════

// Take Profit + Stop Loss (separate settings for Long and Short)
if strategy.position_size > 0
    longSL = strategy.position_avg_price - slMultLong * atrValue
    longTP = strategy.position_avg_price + tpMultLong * atrValue
    strategy.exit("TP/SL Long", "Long", stop=longSL, limit=longTP)

if strategy.position_size < 0
    shortSL = strategy.position_avg_price + slMultShort * atrValue
    shortTP = strategy.position_avg_price - tpMultShort * atrValue
    strategy.exit("TP/SL Short", "Short", stop=shortSL, limit=shortTP)

// MOST reversal exits
if mostBearish and strategy.position_size > 0
    strategy.close("Long", comment="MOST Flip")
if mostBullish and strategy.position_size < 0
    strategy.close("Short", comment="MOST Flip")

// ═══════════════════════════════════════════════════
// PLOTS
// ═══════════════════════════════════════════════════
plot(trendEma, "Trend EMA 200", color=color.white, linewidth=1)
plot(basis, "BB Basis", color=color.gray)
p1 = plot(upper, "BB Upper", color=color.new(color.blue, 50))
p2 = plot(lower, "BB Lower", color=color.new(color.blue, 50))
fill(p1, p2, color=color.new(color.blue, 92), title="BB Fill")
plot(MOST, "MOST", color=dir == 1 ? color.green : color.red, linewidth=2)
plot(exMov, "MOST EMA", color=color.orange)

plotshape(buySignal,  "Buy",  shape.triangleup,   location.belowbar, color.green, size=size.small)
plotshape(sellSignal, "Sell", shape.triangledown,  location.abovebar, color.red,   size=size.small)
bgcolor(entryWindowOK ? color.new(color.blue, 95) : na, title="Entry Window")
